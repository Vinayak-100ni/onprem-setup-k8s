Step 1: Deploy Headlamp (without OIDC)
Since you want token-based login, you don't need the OIDC config. Use a simpler values file:
bash# Create namespace
kubectl create namespace headlamp

# Create simple values file (no OIDC)
cat > /home/headlamp-token-values.yaml << EOF
replicaCount: 1

service:
  type: NodePort

resources:
  requests:
    memory: "128Mi"
    cpu: "100m"
  limits:
    memory: "256Mi"
    cpu: "200m"
EOF

# Add helm repo if not already added
helm repo add headlamp https://headlamp-k8s.github.io/headlamp/
helm repo update

# Deploy
helm install headlamp headlamp/headlamp \
  --namespace headlamp \
  -f /home/headlamp-token-values.yaml

Step 2: Get the Headlamp URL
bash# Wait for pod to be running
kubectl get pods -n headlamp -w

# Get the NodePort
kubectl get svc -n headlamp

# Get master node IP
kubectl get nodes -o wide
Your URL:
http://MASTER_NODE_IP:NODE_PORT

Step 3: Create ServiceAccounts for Each Role
Create one ServiceAccount per person or per role:
bash# Create a namespace for each team area if not exists
kubectl create namespace dev       --dry-run=client -o yaml | kubectl apply -f -
kubectl create namespace staging   --dry-run=client -o yaml | kubectl apply -f -
kubectl create namespace production --dry-run=client -o yaml | kubectl apply -f -
bash# Admin (e.g. Vinayak)
kubectl create serviceaccount vinayak -n default

# DevOps team
kubectl create serviceaccount devops-team -n default

# Developer
kubectl create serviceaccount developer -n default

# Viewer (QA/manager)
kubectl create serviceaccount viewer -n default

Step 4: Create RBAC Bindings
Save and apply this full RBAC file:
bashcat > /home/token-rbac.yaml << 'EOF'
# ── Admin: full cluster access ──
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: token-admin-binding
subjects:
- kind: ServiceAccount
  name: vinayak
  namespace: default
roleRef:
  kind: ClusterRole
  name: cluster-admin
  apiGroup: rbac.authorization.k8s.io

---
# ── DevOps: admin across all namespaces ──
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: token-devops-binding
subjects:
- kind: ServiceAccount
  name: devops-team
  namespace: default
roleRef:
  kind: ClusterRole
  name: admin
  apiGroup: rbac.authorization.k8s.io

---
# ── Developer: edit in dev namespace only ──
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: token-developer-binding
  namespace: dev
subjects:
- kind: ServiceAccount
  name: developer
  namespace: default
roleRef:
  kind: ClusterRole
  name: edit
  apiGroup: rbac.authorization.k8s.io

---
# ── Viewer: read-only everywhere ──
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: token-viewer-binding
subjects:
- kind: ServiceAccount
  name: viewer
  namespace: default
roleRef:
  kind: ClusterRole
  name: view
  apiGroup: rbac.authorization.k8s.io
EOF

kubectl apply -f /home/token-rbac.yaml

Step 5: Create Long-Lived Token Secrets
bashcat > /home/token-secrets.yaml << 'EOF'
# Token for vinayak
apiVersion: v1
kind: Secret
metadata:
  name: vinayak-token
  namespace: default
  annotations:
    kubernetes.io/service-account.name: vinayak
type: kubernetes.io/service-account-token

---
# Token for devops-team
apiVersion: v1
kind: Secret
metadata:
  name: devops-team-token
  namespace: default
  annotations:
    kubernetes.io/service-account.name: devops-team
type: kubernetes.io/service-account-token

---
# Token for developer
apiVersion: v1
kind: Secret
metadata:
  name: developer-token
  namespace: default
  annotations:
    kubernetes.io/service-account.name: developer
type: kubernetes.io/service-account-token

---
# Token for viewer
apiVersion: v1
kind: Secret
metadata:
  name: viewer-token
  namespace: default
  annotations:
    kubernetes.io/service-account.name: viewer
type: kubernetes.io/service-account-token
EOF

kubectl apply -f /home/token-secrets.yaml

Step 6: Extract Tokens to Share with Team
bash# Admin token (Vinayak)
echo "=== VINAYAK (Admin) ===" && \
kubectl get secret vinayak-token -n default \
  -o jsonpath='{.data.token}' | base64 -d && echo

# DevOps token
echo "=== DEVOPS TEAM ===" && \
kubectl get secret devops-team-token -n default \
  -o jsonpath='{.data.token}' | base64 -d && echo

# Developer token
echo "=== DEVELOPER ===" && \
kubectl get secret developer-token -n default \
  -o jsonpath='{.data.token}' | base64 -d && echo

# Viewer token
echo "=== VIEWER ===" && \
kubectl get secret viewer-token -n default \
  -o jsonpath='{.data.token}' | base64 -d && echo
Each token is a long string like:
eyJhbGciOiJSUzI1NiIsImtpZCI6Ii...

Step 7: How Team Members Log In
1. Open browser → http://192.168.X.X:NODE_PORT
2. Headlamp shows a token input box
3. Paste your token
4. Click Login → Dashboard opens with your role's access
That's it — no software to install, no config files, just a URL + token.

Step 8: Managing Tokens Going Forward
bash# Revoke someone's access (delete the secret)
kubectl delete secret developer-token -n default

# Recreate if needed
kubectl apply -f /home/token-secrets.yaml

# Check who has what ServiceAccounts
kubectl get serviceaccounts -n default

# Check all token secrets
kubectl get secrets -n default | grep token
